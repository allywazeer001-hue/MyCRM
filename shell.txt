"use client";
import { useEffect, useState } from "react";
import { useRouter, usePathname } from "next/navigation";
import { Sidebar } from "./sidebar";
import { Topbar } from "./topbar";
import { useAuthStore } from "@/store/auth.store";
import { useModulesStore } from "@/store/modules.store";
import { ToastProvider } from "@/components/ui/toast";
import { can } from "@/lib/can";
import { isAuthenticatedSession } from "@/lib/auth-guard";

const ADMIN_ROUTES = ["/settings", "/users", "/admin", "/studio"];

export function DashboardShell({ children }: { children: React.ReactNode }) {
  const router = useRouter();
  const pathname = usePathname();
  const { user, syncAuthState } = useAuthStore();
  const { fetchModules } = useModulesStore();
  const [ready, setReady] = useState(false);
  const [authenticated, setAuthenticated] = useState(false);

  useEffect(() => {
    // 1. Sync Zustand with the actual token state
    syncAuthState();

    // 2. Verify the real token (not the persisted flag)
    const valid = isAuthenticatedSession();
    setAuthenticated(valid);
    setReady(true);
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  useEffect(() => {
    if (!ready) return;

    if (!authenticated) {
      router.push("/login");
      return;
    }
    if ((user as any)?.mustChangePassword && pathname !== "/change-password") {
      router.push("/change-password");
      return;
    }
    const isAdminRoute = ADMIN_ROUTES.some(r => pathname.startsWith(r));
    if (isAdminRoute && !can(user, "view", "admin")) {
      router.push("/dashboard");
      return;
    }
    fetchModules();
  }, [ready, authenticated, user, pathname, router, fetchModules]);

  if (!ready) return null;
  if (!authenticated) return null;

  return (
    <ToastProvider>
      <div className="flex h-screen bg-gray-50 overflow-hidden">
        <Sidebar />
        <div className="flex-1 flex flex-col overflow-hidden">
          <Topbar />
          <main className="flex-1 overflow-y-auto p-6">
            {children}
          </main>
        </div>
      </div>
    </ToastProvider>
  );
}
